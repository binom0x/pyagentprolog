```Prolog
% ================================================================
% Программа агента-исследователя с картиной мира на Prolog
% ================================================================
% Данная программа описывает поведение агента, который может:
%   - перемещаться между комнатами,
%   - подбирать предметы,
%   - запоминать своё местоположение и собранные предметы,
%   - отвечать на вопросы о состоянии мира.
%
% Картина мира хранится в двух динамических фактах:
%   1) agent_state(Position, Inventory) – текущая позиция и список предметов.
%   2) room_contents(Room, Items) – какие предметы лежат в каждой комнате.
%
% Все изменения выполняются через предикат set_state/1, который атомарно
% заменяет факты, обеспечивая целостность модели.
% ================================================================

% ----------------------------------------------------------------
% 1. Инициализация начальной картины мира
% ----------------------------------------------------------------
% Начальное состояние: агент в холле, инвентарь пуст.
init_world :-
    set_state(agent_state(hall, [])),
    set_state(room_contents(hall, [key, map])),
    set_state(room_contents(kitchen, [apple, knife])),
    set_state(room_contents(garden, [shovel, watering_can])),
    writeln('Картина мира инициализирована.').

% ----------------------------------------------------------------
% 2. Вспомогательные предикаты для работы со списками (изменение, поиск)
% ----------------------------------------------------------------
% Удалить первый элемент X из списка List, результат – Rest.
% Используется при подборе предмета.
remove_item(X, [X|Rest], Rest) :- !.
remove_item(X, [Y|Rest], [Y|NewRest]) :-
    remove_item(X, Rest, NewRest).

% Проверить, содержится ли элемент в списке.
member_item(X, [X|_]).
member_item(X, [_|T]) :- member_item(X, T).

% Добавить элемент в начало списка (для инвентаря).
add_item(X, List, [X|List]).

% ----------------------------------------------------------------
% 3. Предикаты для восприятия мира (сенсоры)
% ----------------------------------------------------------------
% Агент может узнать, где он находится.
current_position(Pos) :-
    agent_state(Pos, _).

% Агент может узнать содержимое текущей комнаты.
current_room_contents(Items) :-
    current_position(Room),
    room_contents(Room, Items).

% Агент может проверить, есть ли предмет в инвентаре.
has_item(Item) :-
    agent_state(_, Inventory),
    member_item(Item, Inventory).

% ----------------------------------------------------------------
% 4. Действия агента (эффекторы), изменяющие картину мира
% ----------------------------------------------------------------
% Все действия строят новое состояние и вызывают set_state/1 для атомарного
% обновления соответствующих фактов.

% Действие: переместиться в другую комнату (если такая существует).
% Предполагается, что комната должна быть определена в room_contents.
go(Room) :-
    % Узнаём текущее состояние
    agent_state(OldPos, Inventory),
    % Проверяем, что целевая комната существует (есть факт room_contents)
    room_contents(Room, _),
    % Формируем новый факт agent_state
    set_state(agent_state(Room, Inventory)),
    format('~w переместился из ~w в ~w.~n', [agent, OldPos, Room]).

% Действие: подобрать предмет из текущей комнаты.
% Предмет должен лежать в комнате и ещё не быть в инвентаре.
take(Item) :-
    current_position(Room),
    room_contents(Room, Items),
    member_item(Item, Items),                     % предмет есть в комнате
    agent_state(Pos, Inventory),
    \+ member_item(Item, Inventory),             % и его нет в инвентаре
    % Удаляем предмет из содержимого комнаты
    remove_item(Item, Items, NewRoomItems),
    set_state(room_contents(Room, NewRoomItems)),
    % Добавляем предмет в инвентарь
    add_item(Item, Inventory, NewInventory),
    set_state(agent_state(Pos, NewInventory)),
    format('~w взял ~w из ~w.~n', [agent, Item, Room]).

% Действие: положить предмет в текущую комнату (из инвентаря).
drop(Item) :-
    has_item(Item),
    current_position(Room),
    agent_state(Pos, Inventory),
    room_contents(Room, Items),
    % Удаляем предмет из инвентаря
    remove_item(Item, Inventory, NewInventory),
    set_state(agent_state(Pos, NewInventory)),
    % Добавляем предмет в комнату
    add_item(Item, Items, NewRoomItems),
    set_state(room_contents(Room, NewRoomItems)),
    format('~w положил ~w в ~w.~n', [agent, Item, Room]).

% ----------------------------------------------------------------
% 5. Запросы к картине мира (предикаты-вопросы)
% ----------------------------------------------------------------
% Вывести полную информацию о текущем состоянии.
show_state :-
    agent_state(Pos, Inventory),
    writeln('========== ТЕКУЩАЯ КАРТИНА МИРА =========='),
    format('Агент в комнате: ~w~n', [Pos]),
    format('Инвентарь: ~w~n', [Inventory]),
    writeln('Содержимое комнат:'),
    forall(room_contents(Room, Items),
           format('   ~w : ~w~n', [Room, Items])),
    writeln('===========================================').

% Подсказка, какие действия доступны в текущей комнате.
what_can_i_do :-
    current_position(Room),
    current_room_contents(Items),
    format('Вы в комнате ~w. Здесь вы можете:~n', [Room]),
    ( Items = [] ->
        format('   - предметов нет~n')
    ;
        forall(member_item(Item, Items),
               format('   - взять ~w~n', [Item]))
    ),
    ( has_item(X) ->
        format('   - положить ~w~n', [X])
    ;
        true
    ),
    format('   - перейти в другую комнату (go(Комната))~n'),
    format('   - показать состояние (show_state)~n').

% ----------------------------------------------------------------
% 6. Планирование простых действий (пример)
% ----------------------------------------------------------------
% Агент хочет иметь определённый предмет. Если его нет в инвентаре,
% он ищет комнату, где этот предмет лежит, идёт туда и берёт.
acquire(Item) :-
    has_item(Item) ->
        format('~w уже имеет ~w.~n', [agent, Item])
    ;
        room_contents(Room, Items),
        member_item(Item, Items) ->
        go(Room),
        take(Item)
    ;
        format('~w нигде нет.~n', [Item]).

% ----------------------------------------------------------------
% 7. Примеры запросов (для тестирования в среде PyAgentProlog)
% ----------------------------------------------------------------
% После загрузки этой программы в агента и вызова init_world, можно выполнять:
%
%   ?- show_state.
%   ?- go(kitchen).
%   ?- take(apple).
%   ?- drop(map).
%   ?- acquire(key).
%   ?- what_can_i_do.
%
% Обратите внимание: все изменения состояния происходят через set_state/1,
% что гарантирует атомарность и согласованность картины мира.
```
